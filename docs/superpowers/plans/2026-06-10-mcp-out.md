# MCP-out Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expose any harness profile as a standards-compliant MCP server via a new `harness-mcp` entry point (stdio + Streamable HTTP transports).

**Architecture:** A new `src/mcp-out/` module wraps the existing one-shot runner (`runProfileOnce`) behind the low-level MCP `Server` from `@modelcontextprotocol/sdk`. Each profile becomes one MCP tool (`run_<slug>`); typed I/O contracts map to the tool's input/output schemas via the existing `src/server/contract.ts` helpers. Tests use the SDK's `InMemoryTransport` with an injected stub runner so no LLM is called.

**Tech Stack:** TypeScript (ESM, `.js` import suffixes), `@modelcontextprotocol/sdk` ^1.29 (already a dependency), zod-parsed profiles, vitest.

**Spec:** `docs/superpowers/specs/2026-06-10-mcp-out-design.md`

## Conventions the engineer must know

- This repo is **ESM with `.js` suffixes on relative imports** even in `.ts` files (`import { x } from '../profiles/schema.js'`).
- No build step needed for dev/tests: `tsx` + vitest run TypeScript directly.
- Run a single test file: `npx vitest run tests/<file>.test.ts`
- Typecheck: `pnpm typecheck`
- `Profile` objects in tests are built with `ProfileSchema.parse(...)` so all defaults are populated. A minimal valid profile needs `name`, `model.{base_url,api_key_env,id}`, `system_prompt`, `artifact_dir`.

// src/mcp-out/tool-spec.ts
import type { Profile } from '../profiles/schema.js'

export interface McpToolSpec {
  name: string
  description: string
  inputSchema: Record<string, unknown>
  outputSchema?: Record<string, unknown>
  /** true when the profile has no input contract — tool input arrives as { prompt } */
  promptFallback: boolean
}

function promptSchema(): Record<string, unknown> {
  return {
    type: 'object',
    properties: { prompt: { type: 'string', description: 'The task or question for the agent' } },
    required: ['prompt'],
  }
}

export function buildMcpToolSpec(profile: Profile): McpToolSpec {
  const slug = profile.name.toLowerCase().replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '')
  const inputContract = profile.contract?.input as Record<string, unknown> | undefined
  return {
    name: `run_${slug}`,
    description: profile.description
      ?? `Run the '${profile.name}' agent and return its final answer.`,
    inputSchema: inputContract ?? promptSchema(),
    outputSchema: profile.contract?.output as Record<string, unknown> | undefined,
    promptFallback: !inputContract,
  }
}
```

import { Client } from '@modelcontextprotocol/sdk/client/index.js'
import { InMemoryTransport } from '@modelcontextprotocol/sdk/inMemory.js'
import { ProfileSchema, type Profile } from '../src/profiles/schema.js'
import { createProfileMcpServer, type ProfileRunner } from '../src/mcp-out/server.js'

    name: 'writer',
    model: { base_url: 'https://api.anthropic.com/v1', api_key_env: 'ANTHROPIC_API_KEY', id: 'claude-sonnet-4-6' },
    system_prompt: './prompts/writer.md',
    artifact_dir: '.harness/writer',

  input: { type: 'object', required: ['topic'], properties: { topic: { type: 'string' } } },
  output: { type: 'object', required: ['draft'], properties: { draft: { type: 'string' } } },

  const [clientTransport, serverTransport] = InMemoryTransport.createLinkedPair()
  const client = new Client({ name: 'test-client', version: '0.0.0' })
  await server.connect(serverTransport)
  await client.connect(clientTransport)
  return client
}

    const { tools } = await client.listTools()

// src/mcp-out/server.ts
import { userInfo } from 'os'
import { Server } from '@modelcontextprotocol/sdk/server/index.js'
import { ListToolsRequestSchema, CallToolRequestSchema } from '@modelcontextprotocol/sdk/types.js'
import type { Profile } from '../profiles/schema.js'
import type { AgentLoopEvents } from '../tools/types.js'
import { compileContract, validateAndBuildInput, processOutput } from '../server/contract.js'
import { runProfileOnce, type RunProfileResult } from '../server/run-profile.js'
import { buildMcpToolSpec } from './tool-spec.js'

export type ProfileRunner = (opts: {
  profile: Profile
  userId: string
  userMessage: string
  events: AgentLoopEvents
}) => Promise<RunProfileResult>

// Headless run: no human attached, so prompt-ACL tools are denied (same as cron runs)
const noopEvents: AgentLoopEvents = {
  onToken: () => {},
  onToolStart: () => {},
  onToolEnd: () => {},
  onToolPrompt: async () => false,
  onSessionEnd: () => {},
  onStats: () => {},
}

function errResult(text: string) {
  return { isError: true, content: [{ type: 'text' as const, text }] }
}

export function createProfileMcpServer(
  profile: Profile,
  runner: ProfileRunner = runProfileOnce
): Server {
  const spec = buildMcpToolSpec(profile)
  const contract = compileContract(profile)
  const server = new Server(
    { name: `harness-${profile.name}`, version: '0.1.0' },
    { capabilities: { tools: {} } }
  )

  server.setRequestHandler(ListToolsRequestSchema, async () => ({
    tools: [{
      name: spec.name,
      description: spec.description,
      inputSchema: spec.inputSchema,
      ...(spec.outputSchema ? { outputSchema: spec.outputSchema } : {}),
    }],
  }))

  server.setRequestHandler(CallToolRequestSchema, async (req) => {
    if (req.params.name !== spec.name) {
      return errResult(`Unknown tool: ${req.params.name}`)
    }
    const args = (req.params.arguments ?? {}) as Record<string, unknown>
    const rawInput = spec.promptFallback ? args.prompt : args
    const input = validateAndBuildInput(rawInput, contract)
    if (!input.ok) {
      const details = input.details ? `\n${input.details.join('\n')}` : ''
      return errResult(`${input.error}${details}`)
    }

    let result: RunProfileResult
    try {
      result = await runner({
        profile,
        userId: process.env.HARNESS_USER ?? userInfo().username,
        userMessage: input.userMessage,
        events: noopEvents,
      })
    } catch (e: any) {
      return errResult(`Agent run failed: ${e.message}`)
    }

    const out = processOutput(result.finalText, contract)
    if (!out.ok) {
      return errResult(`Output contract violation (${out.kind}): ${out.error}`)
    }
    if (typeof out.output === 'string') {
      return { content: [{ type: 'text' as const, text: out.output }] }
    }
    return {
      content: [{ type: 'text' as const, text: JSON.stringify(out.output, null, 2) }],
      structuredContent: out.output as Record<string, unknown>,
    }
  })

  return server
}
```

// src/mcp-out/args.ts
export interface McpArgs {
  profilePath: string
  transport: 'stdio' | 'http'
  port: number
}

export function parseMcpArgs(argv: string[]): McpArgs {
  let profilePath = ''
  let transport: 'stdio' | 'http' = 'stdio'
  let port = 7338
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i]
    if (a === '--transport') {
      const v = argv[++i]
      if (v !== 'stdio' && v !== 'http') throw new Error(`Invalid transport: ${v} (expected stdio|http)`)
      transport = v
    } else if (a === '--port') {
      port = parseInt(argv[++i], 10)
      if (Number.isNaN(port)) throw new Error('Invalid --port (expected a number)')
    } else if (a.startsWith('--')) {
      throw new Error(`Unknown flag: ${a}`)
    } else {
      profilePath = a
    }
  }
  if (!profilePath) {
    throw new Error('Usage: harness-mcp <profile.yaml> [--transport stdio|http] [--port 7338]')
  }
  return { profilePath, transport, port }
}

// src/mcp-out/index.ts
#!/usr/bin/env node
import { resolve } from 'path'
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js'
import { loadProfile } from '../profiles/loader.js'
import { registerAll, clearRegistry, registerTool } from '../tools/builtin/index.js'
import { resolveToolsDirs, loadAllCustomTools } from '../tools/loader.js'
import { createProfileMcpServer } from './server.js'
import { parseMcpArgs } from './args.js'

async function main() {
  const args = parseMcpArgs(process.argv.slice(2))
  const profile = loadProfile(resolve(args.profilePath))

  clearRegistry()
  registerAll()
  const customTools = await loadAllCustomTools(resolveToolsDirs())
  for (const tool of customTools) registerTool(tool)

  if (args.transport === 'http') {
    throw new Error('http transport lands in the next commit — use stdio')
  }

  const server = createProfileMcpServer(profile)
  await server.connect(new StdioServerTransport())
  console.error(`[mcp-out] serving '${profile.name}' on stdio`)
}

main().catch(e => { console.error(e); process.exit(1) })
```

// src/mcp-out/http.ts
import { createServer, type Server as HttpServer } from 'http'
import { StreamableHTTPServerTransport } from '@modelcontextprotocol/sdk/server/streamableHttp.js'
import type { Profile } from '../profiles/schema.js'
import { runProfileOnce } from '../server/run-profile.js'
import { createProfileMcpServer, type ProfileRunner } from './server.js'

export function startHttpMcpServer(
  profile: Profile,
  port: number,
  apiKey?: string,
  runner: ProfileRunner = runProfileOnce
): Promise<HttpServer> {
  const httpServer = createServer((req, res) => {
    void handle(req, res).catch(() => {
      if (!res.headersSent) res.writeHead(500, { 'content-type': 'application/json' })
      res.end(JSON.stringify({ error: 'internal error' }))
    })
  })

  async function handle(req: import('http').IncomingMessage, res: import('http').ServerResponse) {
    if (!req.url?.startsWith('/mcp')) {
      res.writeHead(404, { 'content-type': 'application/json' })
      res.end(JSON.stringify({ error: 'not found' }))
      return
    }
    if (apiKey && req.headers.authorization !== `Bearer ${apiKey}`) {
      res.writeHead(401, { 'content-type': 'application/json' })
      res.end(JSON.stringify({ error: 'unauthorized' }))
      return
    }
    if (req.method !== 'POST') {
      res.writeHead(405, { 'content-type': 'application/json' })
      res.end(JSON.stringify({ error: 'method not allowed' }))
      return
    }
    // Stateless: fresh server + transport per request, torn down with the response
    const server = createProfileMcpServer(profile, runner)
    const transport = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined })
    res.on('close', () => { void transport.close(); void server.close() })
    await server.connect(transport)
    await transport.handleRequest(req, res)
  }

  return new Promise(resolveStart => {
    httpServer.listen(port, () => resolveStart(httpServer))
  })
}
```
