**Goal:** Add `pnpm tui` — a rich Ink-based terminal UI that connects to the same WS server as the readline CLI and shows chat, live tool log, memory keys, token stats, and keybinding hints.

**Tech Stack:** Ink 5, @inkjs/ui, React 19, tsx (already installed), WebSocket (ws — already installed on server; native browser WebSocket API in Node via `ws` on client)

```ts
import { useState, useEffect, useCallback, useRef } from 'react'
import type { ClientMessage, ServerMessage } from '../../types.js'

const HARNESS_PORT = process.env.HARNESS_PORT ?? '7337'

export function parseServerMessage(raw: string): ServerMessage | null {
  try {
    return JSON.parse(raw) as ServerMessage
  } catch {
    return null
  }
}

export interface UseWsResult {
  connected: boolean
  send: (msg: ClientMessage) => void
  lastMessage: ServerMessage | null
}

export function useWs(): UseWsResult {
  const [connected, setConnected] = useState(false)
  const [lastMessage, setLastMessage] = useState<ServerMessage | null>(null)
  const wsRef = useRef<WebSocket | null>(null)
  const reconnectTimer = useRef<ReturnType<typeof setTimeout> | null>(null)

  const connect = useCallback(() => {
    const ws = new WebSocket(`ws://localhost:${HARNESS_PORT}`)
    wsRef.current = ws

    ws.onopen = () => setConnected(true)

    ws.onmessage = (event) => {
      const msg = parseServerMessage(event.data as string)
      if (msg) setLastMessage(msg)
    }

    ws.onclose = () => {
      setConnected(false)
      reconnectTimer.current = setTimeout(connect, 1000)
    }

    ws.onerror = () => {
      // onclose fires after onerror — let it handle reconnect
    }
  }, [])

  useEffect(() => {
    connect()
    return () => {
      if (reconnectTimer.current) clearTimeout(reconnectTimer.current)
      wsRef.current?.close()
    }
  }, [connect])

  const send = useCallback((msg: ClientMessage) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(msg))
    }
  }, [])

  return { connected, send, lastMessage }
}
```

import { useReducer, useCallback, useEffect } from 'react'
import type { ServerMessage } from '../../types.js'

export interface UserMessage {
  role: 'user'
  content: string
  id: string
}

export interface AssistantMessage {
  role: 'assistant'
  content: string
  streaming: boolean
  id: string
}

export interface ToolCallMessage {
  role: 'tool-call'
  name: string
  input: Record<string, unknown>
  result?: string
  status: 'running' | 'ok' | 'err'
  duration_ms?: number
  id: string
}

export interface ToolPromptMessage {
  role: 'tool-prompt'
  name: string
  input: Record<string, unknown>
  id: string
  pending: boolean
}

export type ChatMessage = UserMessage | AssistantMessage | ToolCallMessage | ToolPromptMessage

type Action =
  | { event: 'user'; content: string }
  | { event: 'token'; content: string }
  | { event: 'seal' }
  | { event: 'tool_start'; name: string; input: Record<string, unknown>; id: string }
  | { event: 'tool_end'; name: string; result: string; ok: boolean; duration_ms: number; id: string }
  | { event: 'tool_prompt'; id: string; name: string; input: Record<string, unknown> }
  | { event: 'tool_prompt_resolve'; id: string }
  | { event: 'clear' }

let counter = 0
const uid = () => String(++counter)

export function messagesReducer(state: ChatMessage[], action: Action): ChatMessage[] {
  switch (action.event) {
    case 'user':
      return [...state, { role: 'user', content: action.content, id: uid() }]

    case 'token': {
      const last = state[state.length - 1]
      if (last?.role === 'assistant' && last.streaming) {
        return [
          ...state.slice(0, -1),
          { ...last, content: last.content + action.content },
        ]
      }
      return [...state, { role: 'assistant', content: action.content, streaming: true, id: uid() }]
    }

    case 'seal': {
      const last = state[state.length - 1]
      if (last?.role === 'assistant' && last.streaming) {
        return [...state.slice(0, -1), { ...last, streaming: false }]
      }
      return state
    }

    case 'tool_start':
      return [...state, {
        role: 'tool-call', name: action.name, input: action.input,
        status: 'running', id: action.id,
      }]

    case 'tool_end': {
      return state.map(m =>
        m.role === 'tool-call' && m.name === action.name && m.status === 'running'
          ? { ...m, result: action.result, status: action.ok ? 'ok' : 'err', duration_ms: action.duration_ms }
          : m
      )
    }

    case 'tool_prompt':
      return [...state, {
        role: 'tool-prompt', name: action.name, input: action.input,
        id: action.id, pending: true,
      }]

    case 'tool_prompt_resolve':
      return state.map(m =>
        m.role === 'tool-prompt' && m.id === action.id ? { ...m, pending: false } : m
      )

    case 'clear':
      return []

    default:
      return state
  }
}

export interface UseMessagesResult {
  messages: ChatMessage[]
  dispatch: (action: Action) => void
}

export function useMessages(lastMessage: ServerMessage | null): UseMessagesResult {
  const [messages, dispatch] = useReducer(messagesReducer, [])

  useEffect(() => {
    if (!lastMessage) return
    switch (lastMessage.type) {
      case 'token':
        dispatch({ event: 'token', content: lastMessage.content })
        break
      case 'tool_start':
        dispatch({ event: 'tool_start', name: lastMessage.name, input: lastMessage.input, id: lastMessage.name + Date.now() })
        break
      case 'tool_end':
        dispatch({ event: 'tool_end', name: lastMessage.name, result: lastMessage.result, ok: true, duration_ms: 0, id: lastMessage.name })
        break
      case 'tool_prompt':
        dispatch({ event: 'tool_prompt', id: lastMessage.id, name: lastMessage.name, input: lastMessage.input })
        break
      case 'session_end':
        dispatch({ event: 'seal' })
        break
    }
  }, [lastMessage])

  return { messages, dispatch }
}
```

import { useReducer, useEffect } from 'react'
import type { ServerMessage } from '../../types.js'

export interface ToolEntry {
  name: string
  status: 'running' | 'ok' | 'err'
  duration_ms?: number
  ts: number
}

type Action =
  | { event: 'start'; name: string; input: Record<string, unknown> }
  | { event: 'end'; name: string; ok: boolean; duration_ms: number }

const MAX_ENTRIES = 20

export function toolsReducer(state: ToolEntry[], action: Action): ToolEntry[] {
  switch (action.event) {
    case 'start': {
      const entry: ToolEntry = { name: action.name, status: 'running', ts: Date.now() }
      const next = [...state, entry]
      return next.slice(-MAX_ENTRIES)
    }
    case 'end': {
      // Update most recent running entry for this tool name
      const idx = [...state].reverse().findIndex(e => e.name === action.name && e.status === 'running')
      if (idx === -1) return state
      const realIdx = state.length - 1 - idx
      const updated = [...state]
      updated[realIdx] = {
        ...updated[realIdx],
        status: action.ok ? 'ok' : 'err',
        duration_ms: action.duration_ms,
      }
      return updated
    }
    default:
      return state
  }
}

export function useTools(lastMessage: ServerMessage | null): ToolEntry[] {
  const [tools, dispatch] = useReducer(toolsReducer, [])

  useEffect(() => {
    if (!lastMessage) return
    if (lastMessage.type === 'tool_start') {
      dispatch({ event: 'start', name: lastMessage.name, input: lastMessage.input })
    } else if (lastMessage.type === 'tool_end') {
      const ok = !lastMessage.result.startsWith('Error:')
      dispatch({ event: 'end', name: lastMessage.name, ok, duration_ms: 0 })
    }
  }, [lastMessage])

  return tools
}
```

import { useReducer, useEffect } from 'react'
import type { ServerMessage } from '../../types.js'

export interface MemoryKey {
  key: string
  isNew: boolean
}

type Action =
  | { event: 'write'; key: string }
  | { event: 'age'; key: string }

export function memoryReducer(state: MemoryKey[], action: Action): MemoryKey[] {
  switch (action.event) {
    case 'write': {
      const exists = state.find(k => k.key === action.key)
      if (exists) {
        return state.map(k => k.key === action.key ? { ...k, isNew: true } : k)
      }
      return [...state, { key: action.key, isNew: true }]
    }
    case 'age':
      return state.map(k => k.key === action.key ? { ...k, isNew: false } : k)
    default:
      return state
  }
}

export function useMemory(lastMessage: ServerMessage | null): MemoryKey[] {
  const [keys, dispatch] = useReducer(memoryReducer, [])

  useEffect(() => {
    if (!lastMessage) return
    // MemoryWrite tool_end carries the key as part of the result
    if (
      lastMessage.type === 'tool_end' &&
      lastMessage.name === 'MemoryWrite'
    ) {
      // Result format: "Wrote memory key: <key>"
      const match = lastMessage.result.match(/Wrote memory key:\s*(.+)/)
      if (match) {
        const key = match[1].trim()
        dispatch({ event: 'write', key })
        setTimeout(() => dispatch({ event: 'age', key }), 3000)
      }
    }
  }, [lastMessage])

  return keys
}

import React, { useState, useCallback } from 'react'
import { Box, Text, useInput } from 'ink'
import type { ClientMessage } from '../../types.js'
import type { ChatMessage } from '../hooks/use-messages.js'
import { theme } from '../theme.js'

interface InputBarProps {
  onSend: (msg: ClientMessage) => void
  onClear: () => void
  onScroll: (delta: number) => void
  pendingToolId: string | null
  disabled: boolean
}

const HISTORY_MAX = 50

export function InputBar({ onSend, onClear, onScroll, pendingToolId, disabled }: InputBarProps) {
  const [value, setValue] = useState('')
  const [history, setHistory] = useState<string[]>([])
  const [histIdx, setHistIdx] = useState(-1)

  const submit = useCallback(() => {
    const trimmed = value.trim()
    if (!trimmed) return
    onSend({ type: 'message', content: trimmed })
    setHistory(prev => [trimmed, ...prev].slice(0, HISTORY_MAX))
    setHistIdx(-1)
    setValue('')
  }, [value, onSend])

  useInput((input, key) => {
    if (disabled && !pendingToolId) return

    // Tool prompt keys
    if (pendingToolId) {
      if (input === 'y' || input === 'Y') {
        onSend({ type: 'tool_approve', id: pendingToolId })
        return
      }
      if (input === 'n' || input === 'N' || key.escape) {
        onSend({ type: 'tool_deny', id: pendingToolId })
        return
      }
      return
    }

    if (key.ctrl && input === 'c') {
      onSend({ type: 'cancel' })
      return
    }

    if (key.ctrl && input === 'l') {
      onClear()
      setValue('')
      return
    }

    if (key.upArrow) {
      const nextIdx = Math.min(histIdx + 1, history.length - 1)
      setHistIdx(nextIdx)
      setValue(history[nextIdx] ?? '')
      return
    }

    if (key.downArrow) {
      const nextIdx = Math.max(histIdx - 1, -1)
      setHistIdx(nextIdx)
      setValue(nextIdx === -1 ? '' : history[nextIdx] ?? '')
      return
    }

    if (key.return) {
      submit()
      return
    }

    if (key.backspace || key.delete) {
      setValue(v => v.slice(0, -1))
      return
    }

    if (input && !key.ctrl && !key.meta) {
      setValue(v => v + input)
    }
  })

  const prompt = pendingToolId ? '⚠ y/n ❯ ' : disabled ? '… ❯ ' : '❯ '

  return (
    <Box borderStyle="single" borderColor={theme.border} paddingX={1}>
      <Text color={theme.accent}>{prompt}</Text>
      <Text>{value}</Text>
      <Text color={theme.dim}>▌</Text>
    </Box>
  )
}
